"""
该模块定义了整个应用程序中使用的核心接口，
例如流水线阶段模块的接口。
"""
from abc import ABC, abstractmethod
from typing import Any, Dict, Optional

class StageInterface(ABC):
    """
    抽象基类，定义了流水线阶段模块的契约。
    所有阶段模块都必须实现 `process` 方法。
    """

    @abstractmethod
    def process(self, data: Any, context: Optional[Dict[str, Any]] = None) -> Any:
        """
        处理输入数据并返回结果。

        Orchestrator 会为流水线中的每个阶段调用此方法。

        参数:
            data (Any): 来自前一个阶段的输入数据（或第一个阶段的初始输入）。
                        此数据的类型可能因流水线而异。
            context (Optional[Dict[str, Any]]): 一个可选的字典，包含当前运行中跨阶段共享的上下文信息。
                                               可能包括 'task_id', 'run_logger',（阶段应优先使用此 logger 进行上下文相关的日志记录）
                                                'shared_run_config',
                                               和 'accumulated_metadata'。阶段应只读使用此上下文，
                                               或更新特定约定的键（例如 'accumulated_metadata'）。

        返回:
            Any: 此阶段处理的结果。这将作为输入传递给流水线中的下一个阶段。

        抛出:
            Exception: 允许阶段抛出异常。Orchestrator 将根据其错误策略处理这些异常。
                       鼓励使用特定的自定义异常以实现更好的错误处理。
        """
        raise NotImplementedError("每个阶段模块都必须实现 'process' 方法。")

# 未来可以在此处添加其他核心接口。
