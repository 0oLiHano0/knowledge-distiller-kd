# kd_tool/core/cli_main.py (v4.8 - 初始设计)
# -*- coding: utf-8 -*-

"""
=================================================
c01.cli_main.py.md - KD_Tool 命令行入口 (v4.8)
=================================================

**模块功能**:

- **核心职责**: 提供 `kd_tool` 的命令行接口 (CLI)，作为用户与应用程序交互的总入口。
- **技术选型**: 使用 `Typer` 库构建 CLI，利用其与 Pydantic 和类型提示的良好集成。
- **任务**: 
    1. 解析命令行参数（输入路径、配置文件等）。
    2. 定位并传递配置文件路径给 `ApplicationBuilder`。
    3. 实例化并运行 `ApplicationBuilder` 来启动处理流程。
    4. 提供基本的命令反馈和错误处理。

**规范**:
- **必须**提供一个 `run` 命令来执行默认流水线。
- **必须**允许用户指定输入路径。
- **必须**允许用户指定配置文件路径，并提供合理的默认查找机制。
- **必须**进行顶层异常捕获，向用户显示有意义的错误信息。

---
"""

import typer
from typing import List, Optional
from pathlib import Path
import sys

# 导入核心构建器
from .application_builder import ApplicationBuilder

# 导入 Pydantic 配置加载 (假设在 config.py 中)
from .config import load_config, AppConfig # 假设 load_config 可以处理路径

# ==============================================================================
# Typer 应用实例
# ==============================================================================
# [架构师说明]: 使用 Typer 创建一个 CLI 应用对象。
#               no_args_is_help=True 确保在没有参数时显示帮助信息。
app = typer.Typer(
    name="kd_tool", 
    help="""
    KD_Tool (Knowledge Distiller) v4.0 - 本地化运行的源信息治理工具。
    
    通过多阶段流水线对您的文档进行去重和相似性分析，
    为下游应用 (RAG, 知识库) 提供高质量的个人知识源信息。
    """,
    no_args_is_help=True
)

# ==============================================================================
# 默认配置文件路径
# ==============================================================================
# [架构师说明]: 定义一个默认的配置文件名和可能的查找路径。
DEFAULT_CONFIG_FILENAME = "kd_config.yaml"
# [架构师说明]: 可以考虑在用户主目录、项目根目录等位置查找默认配置。
DEFAULT_CONFIG_PATHS = [
    Path.cwd() / DEFAULT_CONFIG_FILENAME, # 当前工作目录
    Path.home() / ".kd_tool" / DEFAULT_CONFIG_FILENAME, # 用户主目录
]

# ==============================================================================
# 'run' 命令
# ==============================================================================
@app.command(help="运行默认的知识蒸馏处理流水线。")
def run(
    input_paths: List[Path] = typer.Argument(
        ..., # '...' 表示此参数是必需的
        help="""
        要处理的一个或多个输入文件或目录路径。
        **规范**: 必须提供至少一个有效路径。
        """,
        exists=True, # Typer 会自动检查路径是否存在
        file_okay=True, # 允许文件
        dir_okay=True,  # 允许目录
        readable=True # 检查可读性
    ),
    config_file: Optional[Path] = typer.Option(
        None, # 默认为 None，表示使用默认查找机制
        "--config", 
        "-c", 
        help=f"""
        指定要使用的配置文件路径。
        如果未提供，将依次查找: {', '.join(map(str, DEFAULT_CONFIG_PATHS))}
        """,
        exists=True,
        file_okay=True,
        dir_okay=False,
        readable=True
    )
):
    """
    `kd_tool` 的主执行命令。它启动整个处理流水线。
    """
    
    typer.echo("🚀 欢迎使用 KD_Tool v4.0！")
    typer.echo("-" * 30)
    
    # --- 1. 定位配置文件 ---
    actual_config_path: Optional[Path] = config_file
    if actual_config_path is None:
        typer.echo(f"未指定配置文件，尝试在默认位置查找 '{DEFAULT_CONFIG_FILENAME}'...")
        for path in DEFAULT_CONFIG_PATHS:
            if path.is_file():
                actual_config_path = path
                typer.secho(f"✅ 在 {actual_config_path} 找到配置文件。", fg=typer.colors.GREEN)
                break
    
    if actual_config_path is None:
        typer.secho(
            f"❌ 错误: 无法找到配置文件 '{DEFAULT_CONFIG_FILENAME}'，请使用 -c 或 --config 参数指定。", 
            fg=typer.colors.RED, 
            err=True
        )
        raise typer.Exit(code=1) # 异常退出

    # --- 2. 初始化 ApplicationBuilder ---
    typer.echo(f"🔧 使用配置文件 '{actual_config_path}' 初始化应用程序...")
    try:
        builder = ApplicationBuilder(str(actual_config_path))
    except Exception as e:
        typer.secho(f"❌ 错误: 初始化应用程序构建器失败: {e}", fg=typer.colors.RED, err=True)
        # 可以在这里打印更详细的 traceback (如果需要)
        raise typer.Exit(code=2)

    # --- 3. 构建 Application ---
    typer.echo("🏗️ 正在构建应用程序核心组件...")
    try:
        application = builder.build()
        logger = application.logger # 获取已经配置好的 logger
    except Exception as e:
        # 如果 builder 或 logger 尚未成功创建，我们需要一个备用日志或直接打印
        typer.secho(f"❌ 错误: 构建应用程序失败: {e}", fg=typer.colors.RED, err=True)
        raise typer.Exit(code=3)

    # --- 4. 运行流水线 ---
    logger.info("⚡️ 应用程序构建完成，准备启动流水线...")
    try:
        # 将 Path 对象转换为字符串列表传递给 Application
        str_input_paths = [str(p) for p in input_paths]
        application.run_default_pipeline(str_input_paths)
        
        # 检查 Application 内部是否有错误标记 (或者 Orchestrator 返回的结果)
        # 这里假设 Application.run 内部会处理日志输出
        # 我们可以在这里添加一个最终的成功/失败消息

        typer.secho("🎉 流水线执行完成！详情请查看日志。", fg=typer.colors.BRIGHT_GREEN)
        raise typer.Exit(code=0)

    except Exception as e:
        # 这是一个顶层捕获，用于捕获 Orchestrator 或 Application.run 中的意外错误
        logger.critical(f"❌ 在流水线执行过程中发生严重错误: {e}", exc_info=True)
        typer.secho(f"❌ 错误: 流水线执行失败。请查看日志获取详细信息。", fg=typer.colors.RED, err=True)
        raise typer.Exit(code=4)


# ==============================================================================
# 其他命令 (占位符)
# ==============================================================================
@app.command(help="（未来实现）初始化数据库和配置。")
def init():

    raise typer.Exit(code=1)

def show_config(
     config_file: Optional[Path] = typer.Option(None, "--config", "-c", help="指定配置文件路径。")
):
    typer.echo("此功能尚未实现。")
    raise typer.Exit(code=1)

# ==============================================================================
# 运行入口
# ==============================================================================
def main():
    """
    Python 脚本的主入口点。
    **规范**: 仅用于调用 Typer 应用。
    """
    # [架构师说明]: Typer 会自动处理命令行参数的解析和命令分发。
    app()

if __name__ == "__main__":
    # [架构师说明]: 使得此脚本可以直接通过 `python -m kd_tool.core.cli_main` 运行。
    #               或者通过 `setup.py` 的 `entry_points` 配置为 `kd_tool` 命令。
    main()
