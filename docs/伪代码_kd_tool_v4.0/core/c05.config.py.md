"""
该模块定义了应用程序的顶层配置模型 AppConfig。
AppConfig 聚合了所有子模块或组件的配置设置，
这些设置的Pydantic模型定义主要位于 'kd_tool.schemas.settings_models' 模块中。
AppConfig 确保了配置的类型安全、数据验证和结构清晰。
"""

from pydantic import BaseModel, Field, DirectoryPath, FilePath
from typing import Optional, List
from pathlib import Path

# --- 从 schemas.settings_models 模块导入所有需要的配置模型 ---
# 这是所有配置片段（Settings DTOs）的权威来源
from kd_tool.schemas.settings_models import (
    StorageSettingsDTO,
    OrchestratorSettings,
    LoggingSettings,
    CzkawkaPrefilterSettings,
    # 架构说明: AnalysisSettings 内部已聚合了各个 *分析阶段 (Analysis Stage)* 的配置，
    # 并且其内部已更新为使用 *StageSettings 命名。
    AnalysisSettings
)

# --- 应用主配置模型 (AppConfig) ---
class AppConfig(BaseModel):
    """
    应用程序的顶层配置模型。
    它聚合了所有子模块或组件的配置。
    其实例通常在应用程序启动时通过加载外部配置源（如文件、环境变量）来创建。
    """
    project_name: str = Field(
        default="Knowledge Distiller KD-Tool",
        description="项目的名称，可能用于日志、报告或界面显示。"
    )
    project_version: str = Field(
        default="4.0.0-dev", # <-- 更新版本号以反映 v4.0
        description="项目的当前版本号。"
    )

    # --- 核心组件配置 ---
    orchestrator: OrchestratorSettings = Field(
        default_factory=OrchestratorSettings,
        description="Orchestrator（编排器）的配置。"
    )
    storage: StorageSettingsDTO = Field(
        default_factory=StorageSettingsDTO,
        description="存储服务的配置。"
    )
    logging: LoggingSettings = Field(
        default_factory=LoggingSettings,
        description="日志系统的配置。"
    )

    # --- 特定阶段或工具的配置 ---
    prefilter_czkawka: CzkawkaPrefilterSettings = Field(
        default_factory=CzkawkaPrefilterSettings,
        description="Czkawka 预处理阶段（文件级去重工具）的配置。"
    )
    analysis: AnalysisSettings = Field(
        default_factory=AnalysisSettings,
        description="所有分析阶段（MD5, SimHash, Semantic等）的参数配置集合。" #
    )
    
    
    # --- （可选）全局输入/输出路径配置 ---
    # default_input_paths: List[DirectoryPath] = Field(

    #     description="（可选）默认扫描的输入目录路径列表。"
    # )
    # default_output_path: Optional[DirectoryPath] = Field(
    #     default=None,
    #     description="（可选）默认的输出目录路径，用于存放报告、中间文件等。"
    # )

    # --- （可选）全局行为开关 ---
    # debug_mode: bool = Field(
    #     default=False,
    #     description="是否启用应用程序范围的调试模式，可能影响日志级别或某些行为。"
    # )
    # force_reprocess_known_files: bool = Field(
    #     default=False,
    #     description="是否强制重新处理已知的、之前已处理过的文件。"
    # )
    class Config:
        extra = 'forbid'
        validate_assignment = True
        arbitrary_types_allowed = True
