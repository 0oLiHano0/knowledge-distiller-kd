# knowledge_distiller_kd/core/c02.application_builder.py.md (v4.2)

# -*- coding: utf-8 -*-

"""
=================================================
c02.application_builder.py.md - 应用程序构建器伪代码 (v4.2)
=================================================

**模块功能**:

- 加载配置 (`AppConfig`)。
- 初始化日志系统。
- **调用位于各模块内部的工厂**创建核心服务和处理阶段。
- 组装成一个可运行的 `Application` 实例。
- 实现依赖注入 (DI) 的核心逻辑，作为应用程序的 "Composition Root"。

**v4.2 更新**: 
- 引入 PrefilterStage。
- 使用来自 AppConfig 的确定配置路径。
- 导入路径更新为共置工厂模式。

---
"""

# =============================================================================
# Imports (伪代码，仅示意)
# =============================================================================
from typing import Dict, Any, List, Optional

from loguru import Logger

# --- 核心模块导入 ---
# 假设 AppConfig 会聚合所有 Settings DTOs
from kd_tool.core.config import AppConfig 
from kd_tool.core.logging_setup import setup_logging, LoggingSetupError
from kd_tool.core.interfaces import StageInterface
from kd_tool.core.orchestrator import Orchestrator
from kd_tool.storage.storage_interface import StorageInterface
from kd_tool.core.errors import KDToolError 

# --- 工厂导入 (来自新位置) ---
from kd_tool.storage.factory import StorageFactory
from kd_tool.core.orchestrator_factory import OrchestratorFactory
from kd_tool.stages.prefilter.prefilter_factory import PrefilterStageFactory 
from kd_tool.stages.docprocessing.factory import DocumentProcessingStageFactory
from kd_tool.stages.blockmerging.factory import BlockMergerStageFactory 
from kd_tool.stages.md5analysis.factory import MD5AnalysisStageFactory
# ... (未来导入更多 Stage 工厂, 例如 SimHash, Semantic, Decision) ...

# =============================================================================
# Application Container Class 
# =============================================================================
class Application:
    """
    代表一个已构建和配置好的应用程序实例。
    它持有对核心组件的引用，并可以提供一个统一的运行入口。
    """
    def __init__(self,
                 config: AppConfig,
                 logger: Logger,
                 storage: StorageInterface,
                 orchestrator: Orchestrator):
        self.config = config
        self.logger = logger
        self.storage = storage
        self.orchestrator = orchestrator
        self.logger.info("应用程序实例已成功构建。")

    def run_default_pipeline(self, *args, **kwargs) -> Any:
        """
        运行默认配置的主处理流水线。
        """
        self.logger.info("Application: 准备运行默认流水线...")
        try:
            # 实际调用 Orchestrator 的 run 方法
            result = self.orchestrator.run(*args, **kwargs) 
            self.logger.success("Application: 流水线运行成功。")
            return result
        except KDToolError as kde:
            self.logger.exception(f"Application: 流水线运行时发生已知错误: {kde}")
            raise
        except Exception as e:
            self.logger.exception(f"Application: 流水线运行时发生未知顶层错误: {e}")
            raise

# =============================================================================
# Core Build Function 
# =============================================================================
def build_application(config: AppConfig) -> Application:
    """
    根据提供的配置构建并返回一个完整的应用程序实例。
    """
    print("INFO: [builder] 开始构建应用程序 (v4.2)...")

    # --- 步骤 1: 初始化日志系统  ---
    try:
        app_logger = setup_logging(config.logging)
        app_logger.info("日志系统初始化完成。")
    except LoggingSetupError as e:
        print(f"CRITICAL: [builder] 无法初始化日志系统，应用程序无法启动。错误: {e}")
        raise

    # --- 步骤 2: 创建存储服务 ---
    try:
        app_logger.info("正在创建 StorageFactory (from storage/)...")
        storage_factory = StorageFactory(logger=app_logger)
        app_logger.info("正在创建存储服务...")
        storage_service = storage_factory.create(settings=config.storage)
        app_logger.success("存储服务创建成功。")
    except Exception as e:
        app_logger.exception("创建存储服务失败，应用程序无法启动。")
        raise

    # --- 步骤 3: 创建 Stage 工厂 ---
    app_logger.info("正在创建所有 Stage 工厂...")
    prefilter_factory = PrefilterStageFactory(logger=app_logger)
    doc_proc_factory = DocumentProcessingStageFactory(logger=app_logger)
    block_merger_factory = BlockMergerStageFactory(logger=app_logger)
    md5_analysis_factory = MD5AnalysisStageFactory(logger=app_logger)
    # ... 创建其他 Stage 工厂 ...
    app_logger.info("所有 Stage 工厂创建完成。")

    # --- 步骤 4: 创建 Stage 实例 ---
    app_logger.info("正在创建所有 Stage 实例...")
    stage_modules: Dict[str, StageInterface] = {}
    try:
        # 使用 AppConfig 中定义的确定配置路径
        stage_modules["prefilter"] = prefilter_factory.create(
            settings=config.prefilter,
            storage=storage_service
        )
        app_logger.debug("PrefilterStage 创建成功。")

        stage_modules["document_processing"] = doc_proc_factory.create(
            settings=config.document_processing,
            storage=storage_service
        )
        app_logger.debug("DocumentProcessingStage 创建成功。")

        stage_modules["block_merging"] = block_merger_factory.create(
            settings=config.block_merger,
            storage=storage_service
        )
        app_logger.debug("BlockMergerStage 创建成功。")

        stage_modules["md5_analysis"] = md5_analysis_factory.create(
            settings=config.analysis.md5,
            storage=storage_service
        )
        app_logger.debug("MD5AnalysisStage 创建成功。")
        
        # ... 创建其他 Stage 实例 (例如 SimHash, Semantic, Decision) ...

        app_logger.success(f"成功创建 {len(stage_modules)} 个 Stage 实例。")
    except AttributeError as e:
        app_logger.exception(f"创建 Stage 实例失败：配置路径错误或缺失。请检查 AppConfig 结构。错误: {e}")
        raise KDToolError(message=f"配置错误: {e}", original_exception=e) from e
    except Exception as e:
        app_logger.exception("创建 Stage 实例失败，应用程序无法启动。")
        raise KDToolError(message=f"Stage 创建失败: {e}", original_exception=e) from e

    # --- 步骤 5: 创建 Orchestrator ---
    try:
        app_logger.info("正在创建 OrchestratorFactory (from core/)...")
        orchestrator_factory = OrchestratorFactory(logger=app_logger)
        app_logger.info("正在创建 Orchestrator...")
        # 从配置中获取默认顺序
        default_stage_order = config.orchestrator.default_stage_order
        orchestrator = orchestrator_factory.create(
            stage_modules=stage_modules,
            default_stage_order=default_stage_order,
            settings=config.orchestrator
        )
        app_logger.success("Orchestrator 创建成功。")
    except Exception as e:
        app_logger.exception("创建 Orchestrator 失败，应用程序无法启动。")
        raise

    # --- 步骤 6: 创建并返回 Application 实例  ---
    application_instance = Application(
        config=config,
        logger=app_logger,
        storage=storage_service,
        orchestrator=orchestrator
    )
    print("INFO: [builder] 应用程序构建完成。")
    return application_instance
