# kd_tool/core/application_builder.py (v4.5.1 - 集成分析阶段)
# -*- coding: utf-8 -*-

"""
=================================================
c02.application_builder.py.md - 应用程序构建器 (v4.5.1)
=================================================

**模块功能**:

- **核心职责**: 作为应用程序的组合根 (Composition Root)。负责根据配置 (`AppConfig`) 创建和组装应用程序的所有核心组件，
              包括日志系统、存储服务、各个阶段 (Stages) 及其工厂 (Factories)，以及编排器 (Orchestrator)。
- **设计模式**: 实现了构建器模式和依赖注入 (DI)。
- **目标**: 将对象的创建和依赖关系的管理与应用程序的业务逻辑分离，提高模块化和可测试性。

**版本历史**:
- v4.0: 初始设计。
- v4.5.1: 【架构师决策】添加 SimHash 和 Semantic 分析阶段的工厂导入和创建逻辑。

---
"""

from loguru import Logger
from typing import Dict, List

# 导入核心接口和组件
from .interfaces import StorageInterface, StageInterface
from .config import AppConfig, load_config # 假设有 load_config 函数
from .orchestrator import Orchestrator
from .logging_setup import setup_logging

# 导入存储工厂
from ..storage.f03.storage_factory import StorageFactory

# 导入各个 Stage 的工厂
# 【v4.5.1 规范】**必须**在此处导入所有需要的 Stage 工厂。
from ..stages.prefilter.p04.prefilter_factory import PrefilterStageFactory
from ..stages.docprocessing.factory import DocumentProcessingStageFactory
from ..stages.blockmerging.factory import BlockMergerStageFactory
from ..stages.md5analysis.factory import MD5AnalysisStageFactory
from ..stages.simhash_analysis.factory import SimHashAnalysisStageFactory # <-- 新增
from ..stages.semantic_analysis.factory import SemanticAnalysisStageFactory # <-- 新增
from ..stages.decision.factory import DecisionStageFactory # <-- 新增：导入 DecisionStageFactory
from ..stages.cleanup.factory import CleanupStageFactory # <-- 新增：导入 CleanupStageFactory



# 导入编排器工厂
from .f02.orchestrator_factory import OrchestratorFactory

# 定义 Application 类 (可以是简单的数据容器或包含 'run' 方法)
class Application:
    """代表可运行的 KD_Tool 应用程序实例。"""
    def __init__(self, orchestrator: Orchestrator, logger: Logger):
        self.orchestrator = orchestrator
        self.logger = logger

    def run_default_pipeline(self, input_paths: List[str]) -> None:
        """运行默认配置的流水线。"""
        self.logger.info(f"应用程序启动，开始运行默认流水线，输入路径: {input_paths}")
        try:
            # 调用 Orchestrator 执行
            context = self.orchestrator.run(input_paths)
            
            if context.errors:
                self.logger.error("流水线执行完成，但检测到错误：")
                for error in context.errors:
                    self.logger.error(f"- {error}")
            else:
                self.logger.success("流水线执行成功完成！")

        except Exception as e:
            self.logger.critical(f"流水线执行过程中发生未捕获的严重错误: {e}", exc_info=True)


class ApplicationBuilder:
    """
    负责构建 `Application` 实例的构建器。
    **规范**: 这是唯一应该知道如何创建和连接所有组件的地方。
    """

    def __init__(self, config_path: str):
        """
        初始化构建器，并加载配置。
        **参数**:
            config_path (str): 应用程序配置文件的路径。
        """
        self._config: AppConfig = load_config(config_path) # 加载配置
        self._logger: Logger = setup_logging(self._config.logging) # 初始化日志系统
        self._logger.info("ApplicationBuilder 初始化完成，配置已加载，日志系统已设置。")
        self._storage_instance: StorageInterface = None # 缓存存储实例

    def _get_storage_instance(self) -> StorageInterface:
        """
        创建或获取存储服务实例 (单例模式)。
        **规范**: 确保整个应用程序使用同一个存储实例。
        """
        if self._storage_instance is None:
            self._logger.debug("正在创建存储服务实例...")
            storage_factory = StorageFactory(self._logger)
            self._storage_instance = storage_factory.create(self. _config.storage)
            self._logger.info(f"存储服务实例 ({self._config.storage.backend_type}) 创建成功。")
        return self._storage_instance

    def _create_stages(self) -> Dict[str, StageInterface]:
        """
        使用各自的工厂创建所有已配置的 Stage 实例。
        **规范**: 
        - 每个 Stage 的创建逻辑封装在其 Factory 中。
        - Builder 负责调用这些 Factory 并提供必要的依赖 (Storage, Settings)。
        - 返回的字典键名 **必须** 与 `OrchestratorSettings.default_stage_order` 中的名称匹配。
        """
        self._logger.debug("开始创建所有 Stage 实例...")
        storage = self._get_storage_instance()
        stages: Dict[str, StageInterface] = {}

        # 1. Prefilter Stage
        if self._config.prefilter.enabled:
            factory = PrefilterStageFactory(self._logger)
            stages["prefilter"] = factory.create(storage, self._config.prefilter)
            self._logger.debug("PrefilterStage 创建成功。")

        # 2. Document Processing Stage
        if self._config.document_processing.enabled:
            factory = DocumentProcessingStageFactory(self._logger)
            stages["document_processing"] = factory.create(storage, self._config.document_processing)
            self._logger.debug("DocumentProcessingStage 创建成功。")

        # 3. Block Merging Stage
        if self._config.block_merging.enabled:
            factory = BlockMergerStageFactory(self._logger)
            stages["block_merging"] = factory.create(storage, self._config.block_merging)
            self._logger.debug("BlockMergerStage 创建成功。")

        # 4. MD5 Analysis Stage
        if self._config.analysis.md5.enabled:
            factory = MD5AnalysisStageFactory(self._logger)
            stages["md5_analysis"] = factory.create(storage, self._config.analysis.md5)
            self._logger.debug("MD5AnalysisStage 创建成功。")

        # 5. SimHash Analysis Stage 【v4.5.1 新增】
        if self._config.analysis.simhash.enabled:
            factory = SimHashAnalysisStageFactory(self._logger)
            stages["simhash_analysis"] = factory.create(storage, self._config.analysis.simhash)
            self._logger.debug("SimHashAnalysisStage 创建成功。")

        # 6. Semantic Analysis Stage 【v4.5.1 新增】
        if self._config.analysis.semantic.enabled:
            factory = SemanticAnalysisStageFactory(self._logger)
            stages["semantic_analysis"] = factory.create(storage, self._config.analysis.semantic)
            self._logger.debug("SemanticAnalysisStage 创建成功。")

        # 7. Decision Stage 【v4.6 新增】
        if self._config.decision.enabled:
            factory = DecisionStageFactory(self._logger)
            stages["decision"] = factory.create(storage, self._config.decision)
            self._logger.debug("DecisionStage 创建成功。")

        # 8. Cleanup Stage 【v4.7 新增】
        if self._config.cleanup.enabled:
            factory = CleanupStageFactory(self._logger)
            stages["cleanup"] = factory.create(storage, self._config.cleanup)
            self._logger.debug("CleanupStage 创建成功。")

        self._logger.info(f"成功创建 {len(stages)} 个 Stage 实例。")
        return stages

    def build(self) -> Application:
        """
        构建并返回完整的 `Application` 实例。
        **流程**:
        1. 获取存储实例。
        2. 创建所有 Stage 实例。
        3. 创建 Orchestrator 实例，并注入 Stages 和配置。
        4. 创建 Application 实例。
        """
        self._logger.info("开始构建 Application 实例...")
        
        storage = self._get_storage_instance()
        all_stages = self._create_stages()

        # 创建 Orchestrator
        orchestrator_factory = OrchestratorFactory(self._logger)
        orchestrator = orchestrator_factory.create(
            all_stages=all_stages,
            storage=storage, # Orchestrator 需要 Storage 来创建 PipelineContext
            settings=self._config.orchestrator,
            logging_settings=self._config.logging # 传递日志配置给 Context 创建
        )
        self._logger.info("Orchestrator 实例创建成功。")

        # 创建 Application
        app = Application(orchestrator, self._logger)
        self._logger.success("Application 实例构建成功！")
        
        return app
