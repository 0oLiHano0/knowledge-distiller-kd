# knowledge_distiller_kd/core/application_builder.py
"""
此模块负责应用程序的构建和组装过程。

职责:
- 加载或接收应用程序配置 (`AppConfig`)。
- 调用 `setup_logging` 初始化日志系统。
- 调用相关工厂创建和配置核心服务（如存储服务）。
- (未来) 调用相关工厂创建和配置所有处理阶段 (`StageInterface`)。
- 调用 OrchestratorFactory 创建和配置 `Orchestrator`。
- 将所有核心组件组装成一个可运行的 `Application` 实例。
- 实现依赖注入 (DI) 的核心逻辑。

这本质上是应用程序的 "Composition Root"（组合根）。
"""

from typing import Dict, Any, List, Optional

# --- 第三方库导入 ---
from loguru import Logger

# --- 项目内部模块导入 ---
from knowledge_distiller_kd.core.config import AppConfig #
from knowledge_distiller_kd.core.logging_setup import setup_logging, LoggingSetupError
from knowledge_distiller_kd.core.interfaces import StageInterface #
from knowledge_distiller_kd.core.orchestrator import Orchestrator #
from knowledge_distiller_kd.storage.storage_interface import StorageInterface #
from knowledge_distiller_kd.factories.storage_factory import create_storage_service #
from knowledge_distiller_kd.factories.orchestrator_factory import OrchestratorFactory 

# --- 应用程序容器类 ---
class Application:
    """
    代表一个已构建和配置好的应用程序实例。
    它持有对核心组件的引用，并可以提供一个统一的运行入口。
    """
    def __init__(self,
                 config: AppConfig,
                 logger: Logger,
                 storage: StorageInterface,
                 orchestrator: Orchestrator):
        self.config = config
        self.logger = logger
        self.storage = storage
        self.orchestrator = orchestrator
        self.logger.info("应用程序实例已成功构建。")

    def run_default_pipeline(self, input_data: Any, **kwargs) -> Any:
        """
        (示例) 运行默认配置的主处理流水线。
        这是一个方便的方法，实际应用中可能有更复杂的启动逻辑。
        """
        self.logger.info("Application: 准备运行默认流水线...")
        try:
            # 实际调用 Orchestrator 的 run 方法
            result = self.orchestrator.run(input_data=input_data, **kwargs)
            self.logger.info("Application: 流水线运行完成。")
            return result
        except Exception as e:
            self.logger.exception(f"Application: 流水线运行时发生顶层错误: {e}")
            # 根据需要决定是否重新抛出或返回错误状态
            raise

# --- 核心构建函数 ---
def build_application(config: AppConfig) -> Application:
    """
    根据提供的配置构建并返回一个完整的应用程序实例。

    这是应用程序启动时依赖注入流程的核心实现点。

    参数:
        config (AppConfig): 应用程序的完整配置。

    返回:
        Application: 一个准备好运行的应用程序实例。

    可能抛出的异常:
        Exception: 如果在构建过程中发生任何无法恢复的严重错误。
    """
    print("INFO: [builder] 开始构建应用程序...")

    # --- 步骤 1: 初始化日志系统 ---
    try:
        app_logger = setup_logging(config.logging) #
        app_logger.info("日志系统初始化完成。")
    except LoggingSetupError as e:
        print(f"CRITICAL: [builder] 无法初始化日志系统，应用程序无法启动。错误: {e}")
        raise

	# --- 步骤 2: 创建存储服务 ---
    try:
        app_logger.info(f"正在创建 StorageFactory...")
        storage_factory = StorageFactory(logger=app_logger) # 创建工厂实例，注入 Logger
        app_logger.info(f"正在创建存储服务，后端类型: {config.storage.backend_type}...")
        storage_service = storage_factory.create(settings=config.storage) # 使用工厂的 create 方法
        app_logger.info("存储服务创建成功。")
    except Exception as e:
        app_logger.exception("创建存储服务失败，应用程序无法启动。")
        raise

    # --- 步骤 3: 创建各个阶段模块 (Stages) ---
    # 【占位符 - 关键扩展点】
    app_logger.info("正在创建阶段模块...(当前为占位符)")
    stage_modules: Dict[str, StageInterface] = {}
    default_stage_order: List[str] = []
    app_logger.info("阶段模块创建完成 (当前为占位符)。")

    # --- 步骤 3b: 创建 Orchestrator Factory ---
    try:
        app_logger.info("正在创建 OrchestratorFactory...")
        # 导入 OrchestratorFactory
        from knowledge_distiller_kd.factories.orchestrator_factory import OrchestratorFactory 
        orchestrator_factory = OrchestratorFactory(logger=app_logger) # 创建工厂实例，注入 Logger
        app_logger.info("OrchestratorFactory 创建成功。")
    except Exception as e:
        app_logger.exception("创建 OrchestratorFactory 失败，应用程序无法启动。")
        raise

    # --- 步骤 4: 创建 Orchestrator ---
    try:
        app_logger.info("正在创建 Orchestrator...")
        orchestrator = orchestrator_factory.create( # 使用工厂的 create 方法
            stage_modules=stage_modules,          # 注入阶段模块字典
            default_stage_order=default_stage_order, # 注入默认顺序
            settings=config.orchestrator          # 注入 Orchestrator 配置
        )
        app_logger.info("Orchestrator 创建成功。")
    except Exception as e:
        app_logger.exception("创建 Orchestrator 失败，应用程序无法启动。")
        raise

    # --- 步骤 5: 创建并返回 Application 实例 ---
    application_instance = Application(
        config=config,
        logger=app_logger,
        storage=storage_service,
        orchestrator=orchestrator
    )
    print("INFO: [builder] 应用程序构建完成。")
    return application_instance
