# kd_tool/core/orchestrator.py
"""
该模块定义了 Orchestrator 类，它是流水线调度引擎的核心。
负责根据配置和输入数据，按顺序执行各个处理阶段。
"""
from typing import Any, Dict, List, Optional 
import uuid
import time
import traceback
from collections import Counter # 用于检测列表中的重复项

# Logger 类型将从项目中统一的日志模块获取/定义 (例如 kd_tool.core.logging_setup.AppLogger)
# 当前为了类型提示的便利性，我们直接使用 loguru.Logger，但实际获取 Logger 实例应通过统一的日志模块。
from loguru import Logger 

# --- 从 schemas 模块导入核心数据结构 ---
from kd_tool.schemas.settings_models import OrchestratorSettings
from kd_tool.schemas.dtos import OutputResult
# --- 从 core.interfaces 模块导入核心接口 ---
from kd_tool.core.interfaces import StageInterface
from kd_tool.core.errors import KDToolError # 导入KDToolError，位置为kd_tool/core/errors.py

# == Orchestrator自身错误 ==
class OrchestratorError(KDToolError):
    """
    Orchestrator 操作相关的基本异常。
    继承自 KDToolError，以融入应用统一的异常体系。
    """
    def __init__(self, message: str, original_exception: Optional[Exception] = None, **kwargs: Any):
        # 调用 KDToolError 的构造函数，并添加一个特定的上下文信息 'module'
        super().__init__(message, original_exception=original_exception, module="orchestrator", **kwargs)

class OrchestratorSettingsError(OrchestratorError):
    """当提供的 OrchestratorSettings 对象本身不符合要求时抛出（主要由Pydantic处理）。"""
    def __init__(self, message: str, original_exception: Optional[Exception] = None, setting_key: Optional[str] = None, **kwargs: Any):
        # original_exception 可能是一个 Pydantic ValidationError
        super().__init__(message, original_exception=original_exception, setting_key=setting_key, **kwargs)

class InvalidPipelineDefinitionError(OrchestratorError):
    """当 stage_modules 或 default_stage_order 的结构不合法时抛出。"""
    # 如果只是 message，可以不定义 __init__，父类的 __init__ 会被调用
    # 如果需要传递特定上下文，例如：
    # def __init__(self, message: str, definition_part: Optional[str] = None, **kwargs: Any):
    #     super().__init__(message, definition_part=definition_part, **kwargs)
    pass # 保持简洁，除非有特定上下文需要传递

class StageExecutionError(OrchestratorError):
    """当某个阶段模块在执行 process() 方法时发生未捕获的异常时，Orchestrator 可以包装并抛出此错误。"""
    def __init__(self, stage_name: str, original_exception: Exception, details: Optional[str] = None, **kwargs: Any):
        message = f"阶段 '{stage_name}' 执行失败。"
        if details:
            message += f" 详情: {details}"

        # 将 stage_name 和 details 作为上下文信息传递给 KDToolError (通过 OrchestratorError 间接传递)
        # original_exception 也标准地传递
        super().__init__(message, original_exception=original_exception, stage_name=stage_name, details=details, **kwargs)
        # self.stage_name 和 self.original_exception 的显式赋值不再需要，
        # 因为这些信息已由 KDToolError 的 context_info 和 original_exception 属性管理。

class OrchestratorRuntimeError(OrchestratorError):
    """Orchestrator 内部（非阶段执行）的运行时错误。"""
    def __init__(self, message: str, original_exception: Optional[Exception] = None, **kwargs: Any):
        super().__init__(message, original_exception=original_exception, **kwargs)

# OrchestratorSettings Pydantic 模型定义已移至 kd_tool.schemas.settings_models
# OutputResult Pydantic 模型定义已移至 kd_tool.schemas.dtos

# StageInterface 定义已移至 kd_tool.core.interfaces

class Orchestrator:
    """
    流水线编排器。
    负责管理和执行一系列定义好的处理阶段（Stage）。
    """
    def __init__(self, *,
                 stage_modules: Dict[str, StageInterface],
                 default_stage_order: List[str],
                 settings: OrchestratorSettings,
                 logger: Logger): # logger 实例将由工厂或应用入口通过统一日志模块获取并注入
        """
        构造 Orchestrator 实例。

        参数:
            stage_modules (Dict[str, StageInterface]): 阶段名称到阶段模块实例的字典。
                每个模块实例必须实现 StageInterface。
            default_stage_order (List[str]): 默认的阶段执行顺序列表。
                列表中的每个阶段名称必须在 stage_modules 中有对应的实例。
            settings (OrchestratorSettings): Orchestrator 的配置设置实例。
            logger (Logger): 用于日志记录的 Logger 实例。该实例应通过项目的统一日志模块获取。

        可能抛出的异常:
            TypeError: 如果 settings 或 logger 的类型不正确。
            InvalidPipelineDefinitionError: 如果 stage_modules 或 default_stage_order 的定义不合法。
        """
        # 参数类型校验 (防御性)
        if not isinstance(settings, OrchestratorSettings):
            raise TypeError("参数 'settings' 必须是 OrchestratorSettings 的实例。")
        # 这里的 logger 类型检查，最终会依赖于项目中统一日志模块定义的 Logger 类型
        if not hasattr(logger, 'bind') or not all(hasattr(logger, m) for m in ['info', 'error', 'debug', 'warning', 'exception']):
            raise TypeError("参数 'logger' 必须是一个兼容 Loguru 行为的 Logger 实例 (应通过统一日志模块获取)。")

        # 校验 stage_modules 和 default_stage_order 的结构和一致性
        self._validate_pipeline_definition(stage_modules, default_stage_order)

        self._stage_modules: Dict[str, StageInterface] = stage_modules
        self._default_stage_order: List[str] = default_stage_order
        self._settings: OrchestratorSettings = settings
        self._logger: Logger = logger

        self._logger.info(
            f"Orchestrator 已初始化。默认阶段顺序: {self._default_stage_order}。 "
            f"错误处理策略: {self._settings.on_pipeline_error_policy}。"
        )

    def _validate_pipeline_definition(self, 
                                      stage_modules: Dict[str, StageInterface], 
                                      default_stage_order: List[str]) -> None:
        """
        私有辅助方法，用于校验流水线定义的有效性。
        """
        if not stage_modules or not isinstance(stage_modules, dict):
            raise InvalidPipelineDefinitionError("stage_modules 必须是一个非空的字典。")
        if not default_stage_order or not isinstance(default_stage_order, list):
            raise InvalidPipelineDefinitionError("default_stage_order 必须是一个非空的列表。")

        defined_stages_names = set(stage_modules.keys())
        ordered_stages_names = set(default_stage_order)

        if defined_stages_names != ordered_stages_names:
            missing_in_order = defined_stages_names - ordered_stages_names
            missing_in_modules = ordered_stages_names - defined_stages_names
            error_messages = []
            if missing_in_order:
                error_messages.append(f"阶段模块中已定义但在顺序列表中缺失: {sorted(list(missing_in_order))}")
            if missing_in_modules:
                error_messages.append(f"顺序列表中存在但在阶段模块中未定义: {sorted(list(missing_in_modules))}")
            raise InvalidPipelineDefinitionError("; ".join(error_messages))

        if len(default_stage_order) != len(ordered_stages_names):
            duplicates = [item for item, count in Counter(default_stage_order).items() if count > 1]
            raise InvalidPipelineDefinitionError(f"default_stage_order 中包含重复的阶段: {duplicates}")

        for stage_name in default_stage_order:
            module_instance = stage_modules[stage_name]
            if not isinstance(module_instance, StageInterface):
                raise InvalidPipelineDefinitionError(
                    f"阶段模块 '{stage_name}' (类型: {type(module_instance).__name__}) 未实现预期的 StageInterface。"
                )
            # StageInterface 契约保证了 process 方法的存在和可调用性，
            # 此处可省略对 hasattr(module_instance, 'process') 的检查，除非 StageInterface 定义不严格。

    def _determine_stages_to_run(
        self,
        stages_to_run_override: Optional[List[str]] = None,
        stages_to_skip_override: Optional[List[str]] = None
    ) -> List[str]:
        """
        根据默认顺序和运行时覆盖选项，确定实际要执行的阶段列表。
        优先使用 stages_to_run_override。如果未提供，则从默认顺序中排除 stages_to_skip_override。
        """
        # 校验覆盖参数的类型
        if stages_to_run_override is not None and not isinstance(stages_to_run_override, list):
            raise InvalidPipelineDefinitionError("stages_to_run_override (如果提供) 必须是一个列表。")
        if stages_to_skip_override is not None and not isinstance(stages_to_skip_override, list):
            raise InvalidPipelineDefinitionError("stages_to_skip_override (如果提供) 必须是一个列表。")

        # 检查覆盖参数中的阶段名称是否存在于已定义的阶段模块中
        all_defined_stages = set(self._stage_modules.keys())
        if stages_to_run_override:
            unknown_stages = [s for s in stages_to_run_override if s not in all_defined_stages]
            if unknown_stages:
                raise InvalidPipelineDefinitionError(f"stages_to_run_override 中包含未定义的阶段: {unknown_stages}")
            self._logger.debug(f"使用运行时指定的阶段顺序: {stages_to_run_override}")
            return list(stages_to_run_override) # 返回副本

        # 如果没有 stages_to_run_override，则基于默认顺序和 stages_to_skip_override 计算
        effective_stage_order = list(self._default_stage_order) # 开始时使用默认顺序的副本

        if stages_to_skip_override:
            unknown_stages = [s for s in stages_to_skip_override if s not in all_defined_stages]
            if unknown_stages:
                raise InvalidPipelineDefinitionError(f"stages_to_skip_override 中包含未定义的阶段: {unknown_stages}")
            
            effective_stage_order = [s for s in effective_stage_order if s not in stages_to_skip_override]
            self._logger.debug(f"跳过阶段后的执行顺序: {effective_stage_order}")
        
        return effective_stage_order

    def run(self,
            input_data: Any,
            stages_to_run: Optional[List[str]] = None,
            stages_to_skip: Optional[List[str]] = None,
            initial_context: Optional[Dict[str, Any]] = None
            ) -> OutputResult:
        """
        执行完整处理流水线。

        参数:
            input_data: 流水线的初始输入数据。
            stages_to_run: (可选) 覆盖默认阶段顺序，只运行指定的阶段列表。
            stages_to_skip: (可选) 从默认阶段顺序中跳过指定的阶段列表。
                           如果同时提供了 stages_to_run，则此参数被忽略。
            initial_context: (可选) 提供给流水线上下文的初始键值对。这些值将被合并到
                             Orchestrator 创建的运行上下文中，但不会覆盖核心上下文键
                             （如 'task_id', 'run_logger'）。

        返回:
            OutputResult: 包含流水线执行结果的对象。
        """
        task_id = f"{self._settings.default_task_id_prefix}{uuid.uuid4()}"
        # 为本次运行绑定特定的 logger，包含 task_id，便于追踪
        run_logger = self._logger.bind(task_id=task_id, component="OrchestratorRun")

        # 构建运行上下文 (Context)
        # context 对象会在各个阶段之间传递，允许阶段共享信息或访问公共资源。
        context: Dict[str, Any] = {
            'task_id': task_id,
            'run_logger': run_logger, # 阶段模块应使用此 logger 进行日志记录
            'shared_data': {},       # 用于阶段间共享可变数据 (应谨慎使用以避免紧耦合)
            'accumulated_results': {} # 用于阶段累积结果或元数据
            # 未来可以考虑加入 AppConfig 的只读副本或特定部分到 context 中，
            # 如果阶段模块需要访问全局配置但又不想通过构造函数注入整个 AppConfig。
            # 'app_config_snapshot': self._settings_provider.get_read_only_config_view() # 概念性
        }
        if initial_context:
            for key, value in initial_context.items():
                if key in context: # 防止覆盖核心上下文键
                    run_logger.warning(
                        f"初始上下文中提供的键 '{key}' 与 Orchestrator 核心上下文键冲突，将被忽略。"
                    )
                else:
                    context[key] = value
        
        run_logger.info(f"流水线运行启动。输入数据类型: {type(input_data).__name__}。")
        total_start_time = time.monotonic() # 使用 monotonic 以避免系统时间调整影响耗时计算
        stage_durations: Dict[str, float] = {}
        current_data: Any = input_data
        executed_stages_in_order: List[str] = [] # 记录实际执行的阶段及其顺序

        try:
            active_stages = self._determine_stages_to_run(
                stages_to_run_override=stages_to_run,
                stages_to_skip_override=stages_to_skip
            )
            run_logger.info(f"计划执行的阶段顺序: {active_stages}")

            if not active_stages:
                run_logger.warning("没有配置或选择任何阶段来执行。流水线提前结束。")
                # 这种情况下，流水线技术上没有失败，但可能也不是用户期望的。
                # OutputResult 的 message 字段可以用来传达这种情况。
                # ... (返回 OutputResult 的逻辑保持，但可以调整 message) ...
                return OutputResult( # (此部分为简化，具体字段参考之前详细设计)
                    success=True, data=current_data, metadata={'task_id': task_id, 'message': "无阶段执行"},
                    error_message="没有配置或选择任何阶段来执行。", error_stage=None, error_details=None
                )


            for stage_name in active_stages:
                executed_stages_in_order.append(stage_name)
                stage_instance = self._stage_modules[stage_name]
                run_logger.info(f"阶段 '{stage_name}' 开始执行。")
                stage_start_time = time.monotonic()
                
                try:
                    current_data = stage_instance.process(data=current_data, context=context)
                    # 假设 process 方法返回处理后的数据
                except Exception as stage_exc: # 捕获阶段执行期间的任何异常
                    # ... (错误处理逻辑，根据 on_pipeline_error_policy) ...
                    # (此部分为简化，具体错误处理和 OutputResult 构建参考之前详细设计)
                    run_logger.error(f"阶段 '{stage_name}' 失败: {stage_exc}", exc_info=True)
                    if self._settings.on_pipeline_error_policy == 'HALT_ON_FIRST_ERROR':
                        return OutputResult( # (简化)
                            success=False, data=None, error_message=str(stage_exc), error_stage=stage_name,
                            metadata={'task_id': task_id, 'failed_stage': stage_name}
                        )
                    # (如果 CONTINUE_IGNORING_ERROR，则记录错误并继续)
                    # ...
                finally: # 确保记录阶段耗时，无论成功与否
                    stage_duration = time.monotonic() - stage_start_time
                    stage_durations[stage_name] = round(stage_duration, 4)
                    # 检查 stage_exc 是否在当前作用域定义，以判断是否发生异常
                    # (更简洁的方式是在 try 块成功结束时记录，在 except 块中也记录)
                    # 为了简化，我们假设如果 stage_exc 存在，则阶段失败
                    current_stage_failed = 'stage_exc' in locals() and locals()['stage_exc'] is not None
                    if not current_stage_failed:
                         run_logger.info(f"阶段 '{stage_name}' 成功完成。耗时: {stage_duration:.4f} 秒。")
                    if current_stage_failed: # 清理，避免影响下一个循环的判断
                        del locals()['stage_exc']


            # 所有选定阶段执行完毕（或因HALT策略提前中止）
            # ... (构建最终 OutputResult 的逻辑保持，但可以简化以突出框架) ...
            return OutputResult( # (简化)
                success=True, data=current_data, metadata={'task_id': task_id, 'stage_durations': stage_durations}
            )

        except InvalidPipelineDefinitionError as e:
            run_logger.error(f"流水线定义无效，无法执行: {e}", exc_info=False)
            return OutputResult( # (简化)
                success=False, data=None, error_message=str(e), error_stage="PipelineDefinition",
                metadata={'task_id': task_id}
            )
        except Exception as e: # 捕获 Orchestrator 自身在 run 方法内的其他未知错误
            run_logger.exception(f"Orchestrator 在执行 run 方法时发生意外的顶层错误: {e}")
            return OutputResult( # (简化)
                success=False, data=None, error_message=f"Orchestrator 内部运行时错误: {str(e)}",
                error_stage="OrchestratorRuntime", metadata={'task_id': task_id}
            )
        finally: # 确保记录总耗时
            total_duration_seconds = time.monotonic() - total_start_time
            run_logger.info(f"流水线运行结束。总耗时: {total_duration_seconds:.4f} 秒。")
            # 注意：如果 run 方法在 OutputResult 返回前就因异常退出了，
            # total_duration 可能不会被包含在 OutputResult.metadata 中。
            # 之前的详细设计中，OutputResult 的构建更细致，能处理此情况。
            # 这里为了简化框架，暂时忽略了这一点。

    def configure(self, **kwargs) -> None:
        """
        (未来可能实现) 运行时调整 Orchestrator 行为。
        例如：热替换某个阶段模块的实现，或动态修改某些配置参数。
        此功能需要非常谨慎地设计，以管理状态和确保一致性。
        """
        self._logger.warning("Orchestrator.configure() 方法当前未实现。")
        raise NotImplementedError("Orchestrator.configure() 尚未实现。")
