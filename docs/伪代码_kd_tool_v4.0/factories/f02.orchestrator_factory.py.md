# knowledge_distiller_kd/factories/orchestrator_factory.py
"""
此模块定义了 OrchestratorFactory，负责创建 Orchestrator 实例。
它遵循类式工厂模式，确保 Orchestrator 的创建过程被封装，
并正确注入其所有依赖。
"""

from typing import Dict, List
from loguru import Logger

# --- 从 schemas 和 core 模块导入核心定义 ---
from knowledge_distiller_kd.schemas.settings_models import OrchestratorSettings #
from knowledge_distiller_kd.core.interfaces import StageInterface #
from knowledge_distiller_kd.core.orchestrator import Orchestrator #

# --- 工厂相关的自定义异常 (如果需要，可以从 storage_factory 或新的公共工厂异常模块导入) ---
# 暂时假设我们可能需要一个通用的 FactoryError
class FactoryError(Exception):
    """与工厂操作相关的基本异常。"""
    pass

class OrchestratorCreationError(FactoryError):
    """当 Orchestrator 创建失败时抛出。"""
    def __init__(self, message: str, original_exception: Exception):
        super().__init__(f"Orchestrator 创建失败: {message}. 原始错误: {original_exception}")
        self.original_exception = original_exception

# --- 类式工厂定义 ---
class OrchestratorFactory:
    """
    一个类式工厂，负责创建和配置 Orchestrator 实例。
    """
    def __init__(self, logger: Logger):
        """
        初始化工厂。

        参数:
            logger (Logger): 用于日志记录的 Logger 实例。
        """
        self.logger = logger.bind(factory="OrchestratorFactory") # 绑定上下文
        self.logger.info("OrchestratorFactory 实例已创建。")

    def create(self,
               stage_modules: Dict[str, StageInterface],
               default_stage_order: List[str],
               settings: OrchestratorSettings
               ) -> Orchestrator:
        """
        创建并返回一个配置好的 Orchestrator 实例。

        参数:
            stage_modules (Dict[str, StageInterface]): 阶段名称到阶段模块实例的字典。
            default_stage_order (List[str]): 默认的阶段执行顺序列表。
            settings (OrchestratorSettings): Orchestrator 的配置设置实例。

        返回:
            Orchestrator: 一个准备好使用的 Orchestrator 实例。

        可能抛出的异常:
            OrchestratorCreationError: 如果 Orchestrator 初始化失败。
        """
        self.logger.info("尝试创建 Orchestrator 实例...")
        self.logger.debug(f"接收到 {len(stage_modules)} 个阶段模块，默认顺序: {default_stage_order}")
        self.logger.debug(f"使用的 OrchestratorSettings: {settings}")

        try:
            # --- 创建 Orchestrator 实例 ---
            # 我们将从自身 (工厂) 获取的 logger 注入到 Orchestrator 中。
            # 这是关键的 DI 步骤。
            orchestrator_instance = Orchestrator(
                stage_modules=stage_modules,
                default_stage_order=default_stage_order,
                settings=settings,
                logger=self.logger # 注入从工厂获取的 Logger
            )
            self.logger.info("Orchestrator 实例已成功创建。")
            return orchestrator_instance

        except Exception as e:
            # 捕获 Orchestrator 构造函数可能抛出的任何异常
            # (例如，如果阶段定义无效，Orchestrator 的 __init__ 可能会验证并抛出异常)
            self.logger.exception(f"创建 Orchestrator 实例失败。错误: {e}")
            raise OrchestratorCreationError(message=str(e), original_exception=e) from e
