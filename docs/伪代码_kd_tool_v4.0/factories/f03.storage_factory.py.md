# knowledge_distiller_kd/factories/storage_factory.py
"""
此模块定义了 StorageFactory，负责根据配置创建存储服务实例。
它遵循类式工厂模式，以实现项目架构的一致性。
"""

from loguru import Logger

# --- 从 schemas 和 storage 模块导入核心定义 ---
from knowledge_distiller_kd.schemas.settings_models import StorageSettingsDTO #
from knowledge_distiller_kd.storage.storage_interface import StorageInterface #
from knowledge_distiller_kd.storage.sqlite_storage import SQLiteStorage #
# from knowledge_distiller_kd.storage.memory_storage import InMemoryStorage # 当我们实现它时会取消注释

# --- 工厂相关的自定义异常 ---
class FactoryError(Exception):
    """与工厂操作相关的基本异常。"""
    pass

class UnsupportedStorageBackendError(FactoryError):
    """当请求不支持的存储后端类型时抛出。"""
    def __init__(self, backend_type: str):
        super().__init__(f"不支持的存储后端类型: {backend_type}")
        self.backend_type = backend_type

class StorageInitializationError(FactoryError):
    """当存储后端初始化失败时抛出。"""
    def __init__(self, backend_type: str, original_exception: Exception):
        super().__init__(f"存储后端 '{backend_type}' 初始化失败。原始错误: {original_exception}")
        self.backend_type = backend_type
        self.original_exception = original_exception

# --- 类式工厂定义 ---
class StorageFactory:
    """
    一个类式工厂，负责创建和初始化存储服务实例。
    """
    def __init__(self, logger: Logger):
        """
        初始化工厂。

        参数:
            logger (Logger): 用于日志记录的 Logger 实例。工厂本身也需要日志记录。
        """
        self.logger = logger.bind(factory="StorageFactory") # 绑定上下文，便于追踪日志来源
        self.logger.info("StorageFactory 实例已创建。")

    def create(self, settings: StorageSettingsDTO) -> StorageInterface:
        """
        根据提供的设置创建并初始化一个存储服务实例。

        参数:
            settings (StorageSettingsDTO): 存储服务的配置。

        返回:
            StorageInterface: 一个已初始化并准备好使用的存储服务实例。

        可能抛出的异常:
            UnsupportedStorageBackendError: 如果 settings.backend_type 无效。
            StorageInitializationError: 如果所选存储后端初始化失败。
        """
        self.logger.info(f"尝试创建存储服务，后端类型: '{settings.backend_type}'")

        storage_instance: StorageInterface

        # --- 根据后端类型选择实现 ---
        if settings.backend_type == "sqlite":
            self.logger.debug(f"为 'sqlite' 后端实例化 SQLiteStorage...")
            # 注意：我们将工厂的 logger 传递给 SQLiteStorage
            storage_instance = SQLiteStorage(settings=settings, logger=self.logger) #
            
        elif settings.backend_type == "memory_debug":
            self.logger.info("注意: 正在配置 'memory_debug' 存储后端。")
            # from knowledge_distiller_kd.storage.memory_storage import InMemoryStorage
            # storage_instance = InMemoryStorage(settings=settings, logger=self.logger)
            raise NotImplementedError(
                "'memory_debug' (InMemoryStorage) 存储后端当前已规划但尚未实现。"
            )

        else:
            self.logger.error(f"在工厂中遇到不支持的存储后端类型: '{settings.backend_type}'")
            raise UnsupportedStorageBackendError(backend_type=settings.backend_type)

        # --- 初始化存储实例 ---
        try:
            self.logger.debug(f"正在初始化存储后端: '{settings.backend_type}'...")
            storage_instance.initialize() # 调用存储实例的初始化方法
            self.logger.info(f"存储后端 '{settings.backend_type}' 已成功初始化。")
        except Exception as e:
            self.logger.exception(f"存储后端 '{settings.backend_type}' 初始化失败。错误详情: {e}")
            raise StorageInitializationError(backend_type=settings.backend_type, original_exception=e) from e
            
        return storage_instance
