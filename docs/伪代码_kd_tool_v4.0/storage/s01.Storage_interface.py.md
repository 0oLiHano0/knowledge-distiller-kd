# knowledge_distiller_kd/storage/storage_interface.py
"""
该模块定义了存储层的抽象接口（契约）。
所有具体的存储实现（例如 SQLiteStorage）都必须遵守此接口。
"""
from abc import ABC, abstractmethod
from typing import Any, Dict, List, Optional

# 导入 Loguru Logger 类型用于类型提示 (或从项目中统一的日志模块导入 Logger 类型)
from loguru import Logger 

# --- 从 schemas 模块导入核心数据结构 ---
# StorageSettingsDTO 定义了存储服务自身的配置
from knowledge_distiller_kd.schemas.settings_models import StorageSettingsDTO
# 其他 DTOs 定义了与存储交互的数据对象的结构
from knowledge_distiller_kd.schemas.dtos import (
    FileRecordDTO,
    ContentBlockDTO,
    AnalysisResultDTO,
    UserDecisionDTO
)

# ==============================
# == 实现 StorageInterface 时的重要约束与指南 ==
# ==============================
#
# 1.  **数据传输对象 (DTOs)**:
#     - 本接口的所有方法签名都明确使用从 `knowledge_distiller_kd.schemas` 目录导入的 Pydantic DTO 类型。
#     - 具体实现类在与数据库或其他后端交互时，负责将这些 DTOs 与其内部的持久化模型（例如 SQLAlchemy ORM 对象）进行转换。
#     - 输入的 DTOs 应被视为不可变数据，不应在存储实现内部直接修改其内容。如果需要修改，应创建新的 DTO 实例或更新持久化模型后重新生成 DTO。
#
# 2.  **错误处理**:
#     - 实现类必须捕获其操作过程中可能发生的底层异常（如 `sqlalchemy.exc.SQLAlchemyError`, `IOError`, `FileNotFoundError` 等）。
#     - 捕获到的底层异常应被包装成一个通用的、定义在项目核心错误模块（例如 `knowledge_distiller_kd.core.exceptions`）中的自定义存储异常
#       （例如，可以定义 `StorageError(KDToolError)` 基类，以及更具体的 `StorageConnectionError`, `RecordNotFoundError`, `DuplicateRecordError` 等子类）。
#     - 向上层调用者抛出这些自定义的存储异常，以便调用者可以进行统一和明确的错误处理。
#
# 3.  **配置与初始化 (`__init__`, `initialize`)**:
#     - 构造函数 `__init__` 接收 `StorageSettingsDTO` 和 `Logger` 实例，用于依赖注入。构造函数本身不应执行耗时的I/O操作或连接建立。
#     - `initialize()` 方法负责执行实际的后端准备工作，如建立数据库连接、验证/创建必要的表结构（通常通过Alembic迁移完成，但initialize可以做检查）、
#       创建必要的目录（对于文件系统后端）等。此方法在存储实例被工厂创建后、首次使用前调用。
#
# 4.  **清理与资源释放 (`finalize`)**:
#     - `finalize()` 方法负责执行必要的清理任务，如关闭数据库连接、刷新任何挂起的写入操作、释放文件句柄等。
#     - 此方法应在应用程序关闭或存储实例不再需要时被调用，以确保资源被正确释放。
#
# 5.  **幂等性 (Idempotency)**:
#     - `initialize()` 和 `finalize()` 方法的实现应力求幂等。即，多次调用 `initialize()` （如果后端已初始化）不应产生错误或副作用，
#       多次调用 `finalize()` （如果资源已释放）同样如此。
#
# 6.  **原子性与事务 (Atomicity & Transactions)**:
#     - 对于支持事务的后端（如关系型数据库），关键的写操作（例如 `save_content_blocks`, `save_analysis_results`, `save_user_decision`，
#       特别是当它们涉及多个数据库记录的修改时）应该在事务的保护下执行，以保证数据的一致性和完整性。
#     - 接口中定义的 `begin_transaction()`, `commit_transaction()`, `rollback_transaction()` 方法为此提供了显式的事务控制能力。
#       实现类应确保这些方法能正确管理其后端的事务状态。
#
# 7.  **日志记录 (Logging)**:
#     - 实现类通过构造函数接收 Logger 实例，并在关键操作（如连接、重要的数据读写、错误发生时）使用此 Logger 记录结构化的日志。
#     - 日志信息应包含足够的上下文，例如相关记录的ID、操作类型等，以便于追踪和调试。
#
# 8.  **依赖注入 (Dependency Injection)**:
#     - `StorageInterface` 的具体实例将由一个工厂函数（例如 `create_storage_service`）创建。
#     - 该工厂负责将 `StorageSettingsDTO` 和 `Logger` 注入到存储实例中，并调用 `initialize()`。
#     - 其他需要存储服务的模块（如 Orchestrator、各个阶段模块）将通过依赖注入获得 `StorageInterface` 的实例。
#
# 9.  **路径处理 (Path Handling)**:
#     - `register_file` 方法接收 `file_path: str`。具体实现应在内部尽早将其转换为 `pathlib.Path` 对象进行处理，以利用其提供的路径操作能力和跨平台兼容性。
#       DTO `FileRecordDTO.original_path` 已经是 `Path` 类型。
#
# 10. **查询条件 (`criteria` 参数)**:
#     - 接受 `criteria: Optional[Dict[str, Any]]` 的方法（如 `list_file_records`, `get_analysis_results` 等）应清晰地文档化其支持哪些查询键以及值的类型。
#     - 实现时应注意防止SQL注入（如果直接使用字典构建SQL查询，虽然通过ORM通常是安全的）。对查询条件进行适当的验证和清理。
#
# ==============================

class StorageInterface(ABC):
    """
    定义了知识蒸馏工具中所有存储操作的契约的抽象基类。
    任何具体的存储实现都必须继承此类并实现所有抽象方法。
    """

    @abstractmethod
    def __init__(self, settings: StorageSettingsDTO, logger: Logger):
        """
        存储实现的构造函数。

        依赖项（如特定于此存储实例的配置 `settings` 和日志记录器 `logger`）在此处注入。
        构造函数本身应轻量，不执行耗时的I/O操作或连接建立。
        实际的初始化逻辑应放在 `initialize()` 方法中。

        参数:
            settings (StorageSettingsDTO): 此存储实现的特定配置。
            logger (Logger): 用于日志记录的 Logger 实例。
        """
        self.settings = settings # 存储配置以供后续使用
        self.logger = logger   # 存储 logger 实例
        # raise NotImplementedError # 实现类应调用 super().__init__(settings, logger) 或自行处理

    @abstractmethod
    def initialize(self) -> None:
        """
        设置并准备存储后端以供使用。

        此方法应在存储实例创建后、首次使用前被调用。
        具体实现应包括：建立数据库连接、验证或创建必要的表结构/模式
        （通常由数据库迁移工具如 Alembic 管理，但此方法可做存在性检查）、
        创建必要的目录或文件结构（对于文件系统后端）等。

        此方法应是幂等的。
        """
        raise NotImplementedError

    @abstractmethod
    def finalize(self) -> None:
        """
        执行任何必要的清理任务，确保资源被正确释放。

        当应用程序关闭或存储实例不再需要时调用此方法。
        具体实现应包括：关闭数据库连接、刷新任何挂起的缓冲数据、释放文件句柄等。

        此方法应是幂等的。
        """
        raise NotImplementedError

    # --- 文件记录管理 (File Record Management) ---
    @abstractmethod
    def register_file(self, file_path: str, metadata: Optional[Dict[str, Any]] = None) -> FileRecordDTO:
        """
        注册一个新的文件路径到存储系统中，或获取已注册文件的记录。
        如果基于 file_path 的文件记录已存在，则返回现有记录（可选择是否更新元数据）。

        参数:
            file_path (str): 文件的绝对路径。实现内部应转换为 `pathlib.Path`。
            metadata (Optional[Dict[str, Any]]): 与文件关联的可选元数据。
                                               如果文件已存在且提供了元数据，可能用于更新现有元数据。
        返回:
            FileRecordDTO: 已注册或已存在文件的文件记录 DTO。
        
        可能抛出的异常:
            StorageError (或其子类): 如果注册过程中发生错误。
        """
        raise NotImplementedError

    @abstractmethod
    def get_file_record(self, file_id: str) -> Optional[FileRecordDTO]:
        """
        根据文件的唯一 ID 检索文件记录。

        参数:
            file_id (str): 文件的唯一标识符。
        返回:
            Optional[FileRecordDTO]: 如果找到，则为文件记录 DTO；否则为 None。
        """
        raise NotImplementedError

    @abstractmethod
    def list_file_records(self, criteria: Optional[Dict[str, Any]] = None) -> List[FileRecordDTO]:
        """
        列出所有文件记录，可选地根据提供的条件进行过滤。

        参数:
            criteria (Optional[Dict[str, Any]]): 用于过滤文件记录的条件字典。
                例如: `{'processing_status': ProcessingStatus.PENDING}`
        返回:
            List[FileRecordDTO]: 符合条件的文件记录 DTO 列表。如果无匹配，则为空列表。
        """
        raise NotImplementedError
    
    @abstractmethod
    def update_file_record(self, file_id: str, updates: Dict[str, Any]) -> Optional[FileRecordDTO]:
        """
        更新指定 file_id 的文件记录的特定字段。

        参数:
            file_id (str): 要更新的文件的唯一 ID。
            updates (Dict[str, Any]): 包含要更新的字段及其新值的字典。
                例如: `{'processing_status': ProcessingStatus.ANALYZED, 'metadata.tag': 'processed'}`
                (注意: 对于嵌套元数据的更新，实现可能需要特殊处理)
        返回:
            Optional[FileRecordDTO]: 更新后的文件记录 DTO，如果文件未找到则为 None。
        
        可能抛出的异常:
            StorageError (或其子类): 如果更新过程中发生错误。
            RecordNotFoundError: 如果指定的 file_id 不存在。
        """
        raise NotImplementedError

    # --- 内容块管理 (Content Block Management) ---
    @abstractmethod
    def save_content_blocks(self, file_id: str, blocks_data: List[ContentBlockDTO]) -> List[str]:
        """
        保存或更新与特定文件 ID 关联的一批内容块。

        如果 `ContentBlockDTO` 中的 `block_id` 已存在，则应更新该块。
        如果 `block_id` 不存在或未提供（DTO 通常会自动生成），则创建新块。
        此操作应具有原子性，确保一批块要么全部成功保存/更新，要么全部失败。

        参数:
            file_id (str): 这些内容块所属的源文件的 file_id。
            blocks_data (List[ContentBlockDTO]): 要保存或更新的内容块 DTO 列表。
        返回:
            List[str]: 已成功保存或更新的内容块的 block_id 列表。
        
        可能抛出的异常:
            RecordNotFoundError: 如果提供的 file_id 无效。
            StorageError (或其子类): 如果保存过程中发生错误。
        """
        raise NotImplementedError

    @abstractmethod
    def get_content_block(self, block_id: str) -> Optional[ContentBlockDTO]:
        """
        通过其唯一 ID 检索特定的内容块。

        参数:
            block_id (str): 内容块的唯一标识符。
        返回:
            Optional[ContentBlockDTO]: 如果找到，则为内容块 DTO；否则为 None。
        """
        raise NotImplementedError

    @abstractmethod
    def get_content_blocks_for_file(self, file_id: str, criteria: Optional[Dict[str, Any]] = None) -> List[ContentBlockDTO]:
        """
        检索与给定文件 ID 关联的所有内容块，可选地根据条件进行过滤。

        参数:
            file_id (str): 源文件的唯一 file_id。
            criteria (Optional[Dict[str, Any]]): 用于过滤内容块的条件字典 (例如 `{'block_type': BlockType.CODE_SNIPPET}`).
        返回:
            List[ContentBlockDTO]: 内容块 DTO 列表。
        """
        raise NotImplementedError

    @abstractmethod
    def get_all_content_blocks(self, criteria: Optional[Dict[str, Any]] = None) -> List[ContentBlockDTO]:
        """
        检索存储中的所有内容块，可选地根据条件过滤。
        警告：在大型数据集上应谨慎使用此方法，可能导致性能问题或大量内存消耗。
              考虑分页或更具体的查询。

        参数:
            criteria (Optional[Dict[str, Any]]): 过滤条件。
        返回:
            List[ContentBlockDTO]: 内容块 DTO 列表。
        """
        raise NotImplementedError

    # --- 分析结果管理 (Analysis Result Management) ---
    @abstractmethod
    def save_analysis_results(self, results_data: List[AnalysisResultDTO]) -> None: # 或返回 List[str] of pair_analysis_ids
        """
        保存或更新一批分析结果。

        如果具有相同 `pair_analysis_id` 的分析结果已存在，则应更新该结果。
        否则，创建新的分析结果记录。
        此操作应具有原子性。

        参数:
            results_data (List[AnalysisResultDTO]): 要保存或更新的分析结果 DTO 列表。
        """
        raise NotImplementedError

    @abstractmethod
    def get_analysis_results(self, criteria: Optional[Dict[str, Any]] = None) -> List[AnalysisResultDTO]:
        """
        根据指定的过滤条件检索分析结果。

        参数:
            criteria (Optional[Dict[str, Any]]): 过滤条件。
                例如: `{'block_id_1': 'some_block_id', 'analysis_type': AnalysisType.SEMANTIC}`
                       `{'pair_analysis_id': 'a_specific_pair_id'}`
        返回:
            List[AnalysisResultDTO]: 分析结果 DTO 列表。
        """
        raise NotImplementedError

    # --- 用户决策管理 (User Decision Management) ---
    @abstractmethod
    def save_user_decision(self, decision_data: UserDecisionDTO) -> None: # 或返回 UserDecisionDTO
        """
        保存或更新用户（或系统）针对特定分析结果所做的决策。

        决策通过 `pair_analysis_id` 与唯一的 `AnalysisResultDTO` 关联。
        如果具有相同 `pair_analysis_id` 的决策已存在，则应更新该决策。

        参数:
            decision_data (UserDecisionDTO): 用户决策 DTO。
        """
        raise NotImplementedError

    @abstractmethod
    def get_user_decisions(self, criteria: Optional[Dict[str, Any]] = None) -> List[UserDecisionDTO]:
        """
        根据指定的过滤条件检索用户决策。

        参数:
            criteria (Optional[Dict[str, Any]]): 过滤条件。
                例如: `{'pair_analysis_id': 'a_specific_pair_id'}`
                       `{'decision': DecisionType.KEEP}`
        返回:
            List[UserDecisionDTO]: 用户决策 DTO 列表。
        """
        raise NotImplementedError

    # --- 事务管理 (Transaction Management) ---
    # 这些方法对于需要跨多个存储操作保证原子性的调用者非常重要。
    # 例如，一个阶段模块可能需要先保存一些块，然后保存分析结果，
    # 这一系列操作应该在同一个事务中完成。
    @abstractmethod
    def begin_transaction(self) -> None:
        """
        显式开始一个新的事务（如果后端支持并需要手动管理）。
        如果存储后端使用类似 SQLAlchemy Session 的上下文管理器来自动处理事务，
        此方法可能只是一个空操作或内部标记，实际的事务开始由会话管理。
        """
        raise NotImplementedError

    @abstractmethod
    def commit_transaction(self) -> None:
        """
        提交当前活动的事务。
        如果未显式开始事务或后端自动管理事务，此方法可能无效或抛出错误。
        """
        raise NotImplementedError

    @abstractmethod
    def rollback_transaction(self) -> None:
        """
        回滚当前活动的事务。
        如果未显式开始事务或后端自动管理事务，此方法可能无效或抛出错误。
        """
        raise NotImplementedError
