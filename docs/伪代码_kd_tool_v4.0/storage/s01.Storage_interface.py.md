# kd_tool/storage/storage_interface.py （v4.1）

"""
该模块定义了存储层的抽象接口（契约）。
所有具体的存储实现（例如 SQLiteStorage）都必须遵守此接口。
v4.x 更新: 新增 register_files 批量注册方法。
"""
from abc import ABC, abstractmethod
from typing import Any, Dict, List, Optional
from uuid import UUID # <-- 【新增】导入 UUID

# 导入 Loguru Logger 类型用于类型提示
from loguru import Logger 

# --- 从 schemas 模块导入核心数据结构 ---
from kd_tool.schemas.settings_models import StorageSettingsDTO
# 导入 v4.4 DTOs (包含 UUID task_id)
from kd_tool.schemas.dtos import (
    FileRecordDTO,
    ContentBlockDTO,
    AnalysisResultDTO,
    UserDecisionDTO
)

# ==============================
# == 实现 StorageInterface 时的重要约束与指南 ==
# ==============================
#
# 1.  **数据传输对象 (DTOs)**:
#     - 本接口的所有方法签名都明确使用 Pydantic DTO 类型。
#     - 实现类负责 DTOs 与内部持久化模型（如 SQLAlchemy ORM）的转换。
#     - 输入的 DTOs 应被视为不可变。
#
# 2.  **错误处理**:
#     - 实现类必须捕获底层异常并包装成自定义的存储异常（继承自 KDToolError）。
#
# 3.  **配置与初始化 (`__init__`, `initialize`)**:
#     - `__init__` 用于 DI，`initialize()` 用于实际准备工作（连接、建表检查）。
#
# 4.  **清理与资源释放 (`finalize`)**:
#     - `finalize()` 用于关闭连接、释放资源。
#
# 5.  **幂等性 (Idempotency)**:
#     - `initialize()` 和 `finalize()` 应力求幂等。
#
# 6.  **原子性与事务 (Atomicity & Transactions)**:
#     - 关键写操作应在事务保护下执行。
#     - 接口提供 `begin_transaction()`, `commit_transaction()`, `rollback_transaction()`。
#
# 7.  **日志记录 (Logging)**:
#     - 使用注入的 Logger 记录关键操作和错误。
#
# 8.  **依赖注入 (Dependency Injection)**:
#     - 实例由工厂创建和注入。
#
# 9.  **路径处理 (Path Handling)**:
#     - DTO 中已使用 `Path` 类型。接口输入如果是字符串，内部应转换。
#
# 10. **查询条件 (`criteria` 参数)**:
#     - 明确支持的查询键，并注意实现时的安全性。
#
# ==============================

class StorageInterface(ABC):
    """
    定义了知识蒸馏工具中所有存储操作的契约的抽象基类。
    """

    @abstractmethod
    def __init__(self, settings: StorageSettingsDTO, logger: Logger):
        """构造函数，用于依赖注入。"""
        self.settings = settings 
        self.logger = logger   
        # raise NotImplementedError 

    @abstractmethod
    def initialize(self) -> None:
        """设置并准备存储后端。"""
        raise NotImplementedError

    @abstractmethod
    def finalize(self) -> None:
        """执行清理任务，释放资源。"""
        raise NotImplementedError

    # --- 文件记录管理 (File Record Management) ---

    @abstractmethod
    def register_file(self, file_path: str, task_id: UUID, metadata: Optional[Dict[str, Any]] = None) -> FileRecordDTO:
        """
        【保留】注册单个文件路径，或获取已注册记录。
        [架构师说明]: 此方法可能用于非流水线场景或未来的单文件操作。
                      PrefilterStage 将优先使用 register_files。
        """
        raise NotImplementedError

    # --- 【v4.1 新增】批量注册方法 ---
    @abstractmethod
    def register_files(self, files_data: List[FileRecordDTO]) -> List[FileRecordDTO]:
        """
        批量注册或更新文件记录。

        此方法是 PrefilterStage 的主要接口，负责处理一批 FileRecordDTO。
        它应该在事务中执行，以确保数据的一致性。
        如果记录已存在（基于某种唯一键，如 file_id 或 path+hash），则更新；否则创建。

        参数:
            files_data (List[FileRecordDTO]): 要注册或更新的文件记录 DTO 列表。
        返回:
            List[FileRecordDTO]: 成功注册或更新后的文件记录 DTO 列表。
                                 这很重要，因为存储层可能会添加数据库 ID 或
                                 更新时间戳，上层可能需要这些信息。
        可能抛出的异常:
            StorageError: 如果批量注册过程中发生错误。
        """
        raise NotImplementedError
    # --- 结束新增 ---

    @abstractmethod
    def get_file_record(self, file_id: str) -> Optional[FileRecordDTO]:
        """根据文件的唯一 ID 检索文件记录。"""
        raise NotImplementedError

    @abstractmethod
    def list_file_records(self, criteria: Optional[Dict[str, Any]] = None) -> List[FileRecordDTO]:
        """列出所有文件记录，可选过滤。"""
        raise NotImplementedError
    
    @abstractmethod
    def update_file_record(self, file_id: str, updates: Dict[str, Any]) -> Optional[FileRecordDTO]:
        """更新指定 file_id 的文件记录。"""
        raise NotImplementedError

    # --- 内容块管理 (Content Block Management) ---
    @abstractmethod
    def save_content_blocks(self, blocks_data: List[ContentBlockDTO]) -> List[str]:
        """保存或更新一批内容块。"""
        # [修改] 移除了 file_id 参数，因为 ContentBlockDTO 已包含 file_id 和 task_id
        raise NotImplementedError

    @abstractmethod
    def get_content_block(self, block_id: str) -> Optional[ContentBlockDTO]:
        """通过 ID 检索内容块。"""
        raise NotImplementedError

    @abstractmethod
    def get_content_blocks_for_file(self, file_id: str, criteria: Optional[Dict[str, Any]] = None) -> List[ContentBlockDTO]:
        """检索指定文件的所有内容块。"""
        raise NotImplementedError

    @abstractmethod
    def get_all_content_blocks(self, criteria: Optional[Dict[str, Any]] = None) -> List[ContentBlockDTO]:
        """检索所有内容块（谨慎使用）。"""
        raise NotImplementedError

    # --- 分析结果管理 (Analysis Result Management) ---
    @abstractmethod
    def save_analysis_results(self, results_data: List[AnalysisResultDTO]) -> None:
        """保存或更新一批分析结果。"""
        raise NotImplementedError

    @abstractmethod
    def get_analysis_results(self, criteria: Optional[Dict[str, Any]] = None) -> List[AnalysisResultDTO]:
        """根据条件检索分析结果。"""
        raise NotImplementedError

    # --- 用户决策管理 (User Decision Management) ---
    @abstractmethod
    def save_user_decision(self, decision_data: UserDecisionDTO) -> None:
        """保存或更新用户决策。"""
        raise NotImplementedError

    @abstractmethod
    def get_user_decisions(self, criteria: Optional[Dict[str, Any]] = None) -> List[UserDecisionDTO]:
        """根据条件检索用户决策。"""
        raise NotImplementedError

    # --- 事务管理 (Transaction Management) ---
    @abstractmethod
    def begin_transaction(self) -> None:
        """显式开始一个新的事务。"""
        raise NotImplementedError

    @abstractmethod
    def commit_transaction(self) -> None:
        """提交当前活动的事务。"""
        raise NotImplementedError

    @abstractmethod
    def rollback_transaction(self) -> None:
        """回滚当前活动的事务。"""
        raise NotImplementedError

    #     """批量删除指定 ID 的内容块及其相关数据（如分析结果、决策）。"""
    #     raise NotImplementedError

    # @abstractmethod
    # def delete_file_record_and_related_data(self, file_id: str) -> bool: # 返回是否成功
    #     """删除指定文件记录及其所有相关内容块、分析结果、决策。"""
    #     raise NotImplementedError
