# kd_tool/stages/prefilter/prefilter_factory.py(v4.2)

# -*- coding: utf-8 -*-

"""
=================================================
prefilter_factory.py.md - PrefilterStage 工厂伪代码 (v4.2)
=================================================

**模块功能**:

- 负责创建并组装 PrefilterStage 实例及其依赖。
- **v4.2 更新**: 修正 CzkawkaAdapter 的导入路径以匹配共置原则。

---
"""

from loguru import Logger

# --- 核心模块导入 ---
from kd_tool.core.interfaces import StageInterface, StorageInterface
from kd_tool.schemas.settings_models import PrefilterStageSettings
from kd_tool.core.errors import KDToolError 

# --- Stage 和 Adapter 导入 ---
from kd_tool.stages.prefilter.prefilter_stage import PrefilterStage 
from kd_tool.stages.prefilter.adapter_interface import CzkawkaAdapterInterface
# 【重要修改】从 stages/prefilter/ 导入 CzkawkaAdapter
from kd_tool.stages.prefilter.czkawka_adapter import CzkawkaAdapter 

# --- 自定义异常 ---
class FactoryConfigurationError(KDToolError):
    """当工厂遇到配置问题时抛出。"""
    pass

# --- 类式工厂定义 ---
class PrefilterStageFactory:
    """负责创建 PrefilterStage 实例。"""

    def __init__(self, logger: Logger):
        """
        初始化预过滤阶段工厂。
        """
        self._logger = logger.bind(factory_name="PrefilterStageFactory")
        self._logger.info("PrefilterStageFactory initialized.")

    def create(self,
                 settings: PrefilterStageSettings,
                 storage: StorageInterface
                 ) -> StageInterface:
        """
        创建并返回一个配置好的 PrefilterStage 实例。
        """
        self._logger.info("Creating PrefilterStage instance...")

        if settings.tool == "czkawka" and not settings.czkawka:
            self._logger.error("Czkawka settings are missing, but it's the selected tool. Cannot create CzkawkaAdapter.")
            raise FactoryConfigurationError(
                message="PrefilterStage 配置错误：工具设置为 'czkawka'，但 'czkawka' 配置块缺失。"
            )

        if settings.tool == "czkawka":
             czkawka_adapter: CzkawkaAdapterInterface = CzkawkaAdapter(
                settings=settings.czkawka, 
                logger=self._logger.bind(component="CzkawkaAdapter")
             )
        else:
             raise FactoryConfigurationError(f"不支持的预过滤工具: {settings.tool}")

        stage_instance = PrefilterStage(
            logger=self._logger.bind(stage_name="Prefilter"),
            settings=settings,
            storage=storage,
            czkawka_adapter=czkawka_adapter 
        )

        self._logger.success("PrefilterStage instance created successfully.")
        return stage_instance
