# kd_tool/stages/prefilter/prefilter_stage.py (v4.5)

# -*- coding: utf-8 -*-

"""
=================================================
prefilter_stage.py.md - PrefilterStage 伪代码 (v4.5 - 更新处理状态)
=================================================

**模块功能**:

- 负责执行文件级预过滤（如去重）的阶段。
- **v4.5 更新**: 
    - 根据选项 B 决策，更新 `_register_scan_results` 中设置 `FileRecordDTO` 的
      `processing_status`。
    - 非重复文件（包括原始文件和唯一文件）标记为 `PREPROCESSING_COMPLETED`。
    - 重复文件标记为 `DUPLICATE`。

**架构师说明**:

- **[规范] 状态精确**: Stage 在完成其核心职责后，应尽可能精确地更新 DTO 的
             `processing_status`，以准确反映其处理结果，为下游 Stage
             提供清晰的判断依据。

---
"""

from loguru import Logger
from typing import List, Dict
from pathlib import Path
from uuid import UUID

# --- 核心模块导入 ---
from kd_tool.core.interfaces import StageInterface, StorageInterface
from kd_tool.schemas.settings_models import PrefilterStageSettings
# 导入 v4.4 的 DTOs (包含 UUID task_id 和 Logger)
# 导入 v4.x 的 Enums (包含 DUPLICATE 和 PREPROCESSING_COMPLETED)
from kd_tool.schemas.dtos import PipelineContextDTO, FileRecordDTO, ProcessingStatus
from kd_tool.stages.prefilter.adapter_interface import CzkawkaAdapterInterface, CzkawkaScanOutputDTO
from kd_tool.stages.prefilter.errors import PrefilterError

class PrefilterStage(StageInterface):
    """
    负责执行文件级预过滤（如去重）的阶段。
    """

    def __init__(self,
                 logger: Logger,
                 settings: PrefilterStageSettings,
                 storage: StorageInterface,
                 czkawka_adapter: CzkawkaAdapterInterface):
        self._logger = logger.bind(stage_name=self.__class__.__name__)
        self._settings = settings
        self._storage = storage
        self._adapter = czkawka_adapter
        self._logger.info("PrefilterStage initialized.")

    def process(self, context: PipelineContextDTO) -> PipelineContextDTO:
        """
        执行预过滤流程。
        """
        self._logger.info("Starting prefilter stage...")
        task_id: UUID = context.task_id
        run_logger: Logger = context.run_logger

        if not self._settings.enabled:
            run_logger.warning("PrefilterStage is disabled. Skipping.")
            # 如果跳过，文件状态保持不变 (可能是 PENDING 或其他)
            return context

        if self._settings.tool != "czkawka":
            raise PrefilterError(f"Unsupported prefilter tool: {self._settings.tool}")

        try:
            run_logger.info("Running Czkawka to scan and find duplicates...")
            scan_output: CzkawkaScanOutputDTO = self._adapter.scan_and_find_duplicates()
            run_logger.success(f"Czkawka finished. Scanned {len(scan_output.all_scanned_files)} files. Found {len(scan_output.duplicate_groups)} duplicate groups.")

            file_dtos: List[FileRecordDTO] = []

            if self._settings.register_files_in_storage:
                run_logger.info("Registering scan results in storage...")
                file_dtos = self._register_scan_results(scan_output, task_id, run_logger)
                run_logger.success(f"Scan results registered. Found {len(file_dtos)} files.")
            else:
                 run_logger.warning("Registering files in storage is disabled.")

            if file_dtos:
                run_logger.info(f"Updating PipelineContextDTO with {len(file_dtos)} FileRecordDTOs...")
                context.file_records.update({dto.file_id: dto for dto in file_dtos})
                run_logger.success("PipelineContextDTO updated.")
            else:
                run_logger.info("No new FileRecordDTOs to add to PipelineContextDTO.")

            context.shared_data["prefilter_results"] = {
                "scanned_files_count": len(scan_output.all_scanned_files),
                "duplicate_groups_count": len(scan_output.duplicate_groups),
                "registered_files_count": len(file_dtos)
            }

        except Exception as e:
            run_logger.exception("Error during PrefilterStage execution.")
            # [思考] 发生错误时，是否应该将相关文件的状态标记为 PREPROCESSING_FAILED？
            # 当前是直接抛出异常，由 Orchestrator 决定如何处理。这通常是合理的。
            raise PrefilterError(f"Prefilter execution failed: {e}", original_exception=e) from e

        self._logger.info("Prefilter stage finished.")
        return context

    # 【v4.5 修改】更新 status 设置逻辑
    def _register_scan_results(self, scan_output: CzkawkaScanOutputDTO, task_id: UUID, run_logger: Logger) -> List[FileRecordDTO]:
        """
        将 Czkawka 扫描到的所有文件注册到存储服务，并返回 FileRecordDTO 列表。
        """
        files_to_register: Dict[Path, FileRecordDTO] = {}
        processed_duplicates = set()

        run_logger.debug(f"Processing {len(scan_output.duplicate_groups)} duplicate groups...")
        for group in scan_output.duplicate_groups:
            original_path = group.original_file.resolve()

            if original_path not in files_to_register:
                 files_to_register[original_path] = FileRecordDTO(
                    original_path=original_path,
                    size_bytes=group.size_bytes,
                    # 【v4.5 修改】原始文件完成预处理，进入下一阶段
                    status=ProcessingStatus.PREPROCESSING_COMPLETED, 
                    task_id=task_id,
                    metadata={"is_original_duplicate": True}
                )
            processed_duplicates.add(original_path)

            for dup_path_obj in group.duplicates:
                dup_path = dup_path_obj.resolve()
                if dup_path not in files_to_register:
                    files_to_register[dup_path] = FileRecordDTO(
                        original_path=dup_path,
                        size_bytes=group.size_bytes,
                        # 【v4.5 确认】重复文件标记为 DUPLICATE
                        status=ProcessingStatus.DUPLICATE, 
                        task_id=task_id,
                        metadata={
                            "duplicate_of": str(original_path)
                        }
                    )
                processed_duplicates.add(dup_path)

        run_logger.debug(f"Processed {len(processed_duplicates)} files from duplicate groups.")

        run_logger.debug(f"Processing {len(scan_output.all_scanned_files)} scanned files to find unique ones...")
        for file_path_obj in scan_output.all_scanned_files:
            file_path = file_path_obj.resolve()
            if file_path not in processed_duplicates and file_path not in files_to_register:
                try:
                    size = file_path.stat().st_size
                except OSError as e:
                    size = -1
                    run_logger.warning(f"Could not get size for file: {file_path}. Error: {e}")

                files_to_register[file_path] = FileRecordDTO(
                    original_path=file_path,
                    size_bytes=size,
                    # 【v4.5 修改】唯一文件完成预处理，进入下一阶段
                    status=ProcessingStatus.PREPROCESSING_COMPLETED, 
                    task_id=task_id,
                    metadata={}
                )

        unique_count = len(files_to_register) - len(processed_duplicates)
        run_logger.debug(f"Found {unique_count if unique_count >= 0 else 0} unique files.")

        file_dtos_list = list(files_to_register.values())
        if file_dtos_list:
            run_logger.info(f"Registering a total of {len(file_dtos_list)} files...")
            try:
                with self._storage.begin_transaction():
                    self._storage.register_files(file_dtos_list)
                run_logger.success("Bulk registration complete.")
            except Exception as e:
                run_logger.exception("Failed to register files in storage.")
                raise PrefilterError(f"Failed to register files: {e}", original_exception=e) from e
        else:
            run_logger.warning("No files found to register.")

        return file_dtos_list
